@OptIn(ExperimentalFoundationApi::class)
@Composable
fun AuroCategoryListScreen(
    category: LibraryCategory,
    displayedTracks: List<Track>,
    displayedGroups: List<GroupItem>,
    currentPlayingId: Long,
    onBackClick: () -> Unit,
    onTrackClick: (Track, Int) -> Unit,
    onTrackLongClick: (Track) -> Unit,
    onTrackMenuClick: (Track) -> Unit,
    onGroupClick: (GroupItem) -> Unit,
    onGroupLongClick: (GroupItem) -> Unit = {},
    onPlayAll: () -> Unit = {},
    onShuffleAll: () -> Unit = {},
    onCreatePlaylist: () -> Unit = {},
    listState: androidx.compose.foundation.lazy.LazyListState = rememberLazyListState(),
    onSortClick: () -> Unit,
    errorTrackIds: Set<Long> = emptySet(),
    modifier: Modifier = Modifier,
    isSelectionMode: Boolean = false,
    selectedTrackIds: Set<Long> = emptySet(),
    onToggleSelectTrack: (Track) -> Unit = {},
    showFileFormat: Boolean = true
) {
    val isAllSongs = category == LibraryCategory.ALL_SONGS
    val isPlaylists = category == LibraryCategory.PLAYLISTS

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(AuroColor.Black)
    ) {
        AuroBreadcrumbHeader(
            items = listOf(
                AuroBreadcrumbItem("Biblioteca"),
                AuroBreadcrumbItem(category.title)
            ),
            onItemClick = { index ->
                if (index == 0) onBackClick()
            },
            canGoBack = true,
            onBackClick = onBackClick,
            rightActions = {
                IconButton(onClick = onSortClick, modifier = Modifier.size(34.dp)) {
                    Icon(
                        imageVector = Icons.AutoMirrored.Rounded.Sort,
                        contentDescription = "Ordenar",
                        tint = AuroColor.TextSecondary,
                        modifier = Modifier.size(18.dp)
                    )
                }
            }
        )

        // Banner de Ações Rápidas em "Todas as Músicas"
        if (isAllSongs && displayedTracks.isNotEmpty()) {
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 14.dp, vertical = 8.dp),
                horizontalArrangement = Arrangement.spacedBy(10.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                AuroGlass(
                    kind = AuroGlassKind.Card,
                    onClick = onPlayAll,
                    modifier = Modifier
                        .weight(1f)
                        .height(42.dp)
                ) {
                    Row(
                        modifier = Modifier.fillMaxSize(),
                        horizontalArrangement = Arrangement.Center,
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Icon(
                            imageVector = Icons.Rounded.PlayArrow,
                            contentDescription = null,
                            tint = AuroColor.TextPrimary,
                            modifier = Modifier.size(18.dp)
                        )
                        Spacer(modifier = Modifier.width(6.dp))
                        Text(
                            text = "Tocar Todas",
                            color = AuroColor.TextPrimary,
                            fontSize = 13.sp,
                            fontWeight = FontWeight.Medium
                        )
                    }
                }

                AuroGlass(
                    kind = AuroGlassKind.Card,
                    onClick = onShuffleAll,
                    modifier = Modifier
                        .weight(1f)
                        .height(42.dp)
                ) {
                    Row(
                        modifier = Modifier.fillMaxSize(),
                        horizontalArrangement = Arrangement.Center,
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Icon(
                            imageVector = Icons.Rounded.Shuffle,
                            contentDescription = null,
                            tint = AuroColor.TextPrimary,
                            modifier = Modifier.size(16.dp)
                        )
                        Spacer(modifier = Modifier.width(6.dp))
                        Text(
                            text = "Aleatório",
                            color = AuroColor.TextPrimary,
                            fontSize = 13.sp,
                            fontWeight = FontWeight.Medium
                        )
                    }
                }
            }
            HorizontalDivider(color = AuroColor.Hairline, thickness = 1.dp)
        }

        // Ação de Nova Lista de Reprodução
        if (isPlaylists) {
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .clickable { onCreatePlaylist() }
                    .padding(horizontal = 16.dp, vertical = 12.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                Icon(
                    imageVector = Icons.Rounded.Add,
                    contentDescription = null,
                    tint = LocalAuroAccent.current.a.value,
                    modifier = Modifier.size(20.dp)
                )
                Spacer(modifier = Modifier.width(10.dp))
                Text(
                    text = "+ Nova Lista de Reprodução",
                    color = LocalAuroAccent.current.a.value,
                    fontSize = 14.sp,
                    fontWeight = FontWeight.Bold
                )
            }
            HorizontalDivider(color = AuroColor.Hairline, thickness = 1.dp)
        }

        val coroutineScope = rememberCoroutineScope()

        Box(modifier = Modifier.fillMaxSize()) {
            if (isAllSongs) {
                if (displayedTracks.isEmpty()) {
                    AuroEmptyState(
                        icon = Icons.Rounded.MusicOff,
                        title = stringResource(R.string.empty_library_title),
                        message = stringResource(R.string.empty_library_message),
                        actionLabel = stringResource(R.string.empty_library_action),
                        onAction = onBackClick,
                        modifier = Modifier.padding(bottom = 120.dp)
                    )
                } else {
                    LazyColumn(
                        state = listState,
                        modifier = Modifier.fillMaxSize(),
                        contentPadding = PaddingValues(bottom = 120.dp, end = if (displayedTracks.size > 15) 28.dp else 0.dp)
                    ) {
                        itemsIndexed(displayedTracks, key = { _, t -> t.id }) { index, track ->
                            AuroTrackRow(
                                track = track,
                                trackIndex = index,
                                isPlayingThis = track.id == currentPlayingId,
                                hasError = errorTrackIds.contains(track.id),
                                isSelectionMode = isSelectionMode,
                                isSelected = selectedTrackIds.contains(track.id),
                                showFormatTag = showFileFormat,
                                onClick = {
                                    if (isSelectionMode) {
                                        onToggleSelectTrack(track)
                                    } else {
                                        onTrackClick(track, index)
                                    }
                                },
                                onLongClick = { onTrackLongClick(track) },
                                onMenuClick = { onTrackMenuClick(track) }
                            )
                            HorizontalDivider(color = AuroColor.Hairline, thickness = 1.dp)
                        }
                    }

                    if (displayedTracks.size > 15) {
                        AuroAlphabetFastScroller(
                            modifier = Modifier
                                .align(Alignment.CenterEnd)
                                .padding(bottom = 120.dp),
                            onLetterSelected = { letter ->
                                val targetIndex = if (letter == '#') {
                                    displayedTracks.indexOfFirst { it.title.firstOrNull()?.isLetter() == false }
                                } else {
                                    displayedTracks.indexOfFirst { it.title.startsWith(letter, ignoreCase = true) }
                                }
                                if (targetIndex >= 0) {
                                    coroutineScope.launch { listState.scrollToItem(targetIndex) }
                                }
                            }
                        )
                    }
                }
            } else {
                if (displayedGroups.isEmpty()) {
                    if (isPlaylists) {
                        AuroEmptyState(
                            icon = Icons.AutoMirrored.Rounded.QueueMusic,
                            title = stringResource(R.string.empty_playlist_title),
                            message = stringResource(R.string.empty_playlist_message),
                            actionLabel = stringResource(R.string.create_playlist),
                            onAction = onCreatePlaylist,
                            modifier = Modifier.padding(bottom = 120.dp)
                        )
                    } else {
                        AuroEmptyState(
                            icon = Icons.Rounded.FolderOff,
                            title = stringResource(R.string.empty_library_title),
                            message = stringResource(R.string.empty_library_message),
                            modifier = Modifier.padding(bottom = 120.dp)
                        )
                    }
                } else {
                    LazyColumn(
                        state = listState,
                        modifier = Modifier.fillMaxSize(),
                        contentPadding = PaddingValues(bottom = 120.dp, end = if (displayedGroups.size > 15) 28.dp else 0.dp)
                    ) {
                        items(displayedGroups, key = { it.title }) { group ->
                            Row(
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .combinedClickable(
                                        onClick = { onGroupClick(group) },
                                        onLongClick = { onGroupLongClick(group) }
                                    )
                                    .padding(horizontal = 16.dp, vertical = 14.dp),
                                verticalAlignment = Alignment.CenterVertically
                            ) {
                                Column(modifier = Modifier.weight(1f)) {
                                    Text(
                                        text = group.title,
                                        color = AuroColor.TextPrimary,
                                        fontSize = 15.sp,
                                        fontWeight = FontWeight.Medium,
                                        maxLines = 1,
                                        overflow = TextOverflow.Ellipsis
                                    )
                                    Spacer(modifier = Modifier.height(2.dp))
                                    Text(
                                        text = group.subtitle,
                                        color = AuroColor.TextSecondary,
                                        fontSize = 12.sp
                                    )
                                }
                                Icon(
                                    imageVector = Icons.Rounded.ChevronRight,
                                    contentDescription = null,
                                    tint = AuroColor.TextTertiary,
                                    modifier = Modifier.size(18.dp)
                                )
                            }
                            HorizontalDivider(color = AuroColor.Hairline, thickness = 1.dp)
                        }
                    }

                    if (displayedGroups.size > 15) {
                        AuroAlphabetFastScroller(
                            modifier = Modifier
                                .align(Alignment.CenterEnd)
                                .padding(bottom = 120.dp),
                            onLetterSelected = { letter ->
                                val targetIndex = if (letter == '#') {
                                    displayedGroups.indexOfFirst { it.title.firstOrNull()?.isLetter() == false }
                                } else {
                                    displayedGroups.indexOfFirst { it.title.startsWith(letter, ignoreCase = true) }
                                }
                                if (targetIndex >= 0) {
                                    coroutineScope.launch { listState.scrollToItem(targetIndex) }
                                }
                            }
                        )
                    }
                }
            }
        }
    }
}

@Composable
fun AuroLyricsSheet(
    track: Track,
    lyrics: LyricsManager.Lyrics?,
    currentProgressMs: Long,
    onSeekToMs: (Long) -> Unit,
    onDismiss: () -> Unit
) {
    val listState = rememberLazyListState()
    val accent = LocalAuroAccent.current.a.value

    val activeLineIndex = remember(lyrics, currentProgressMs) {
        if (lyrics != null && lyrics.isSynced && lyrics.lines.isNotEmpty()) {
            val idx = lyrics.lines.indexOfLast { it.timeMs <= currentProgressMs }
            if (idx >= 0) idx else 0
        } else -1
    }

    LaunchedEffect(activeLineIndex) {
        if (activeLineIndex >= 0) {
            val targetScroll = (activeLineIndex - 3).coerceAtLeast(0)
            listState.animateScrollToItem(targetScroll)
        }
    }

    AuroOverlay(
        visible = true,
        onDismiss = onDismiss,
        type = AuroOverlayType.BottomSheet
    ) {
        Surface(
            color = AuroColor.Black,
            shape = RoundedCornerShape(topStart = AuroRadius.card, topEnd = AuroRadius.card),
            modifier = Modifier
                .fillMaxWidth()
                .fillMaxHeight(0.82f)
                .border(1.dp, AuroColor.Hairline, RoundedCornerShape(topStart = AuroRadius.card, topEnd = AuroRadius.card))
                .navigationBarsPadding()
        ) {
            Column(modifier = Modifier.padding(horizontal = 20.dp, vertical = 16.dp)) {
                // Drag handle
                Box(
                    modifier = Modifier
                        .align(Alignment.CenterHorizontally)
                        .width(38.dp)
                        .height(4.dp)
                        .background(AuroColor.Hairline, RoundedCornerShape(2.dp))
                )

                Spacer(modifier = Modifier.height(14.dp))

                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Column(modifier = Modifier.weight(1f)) {
                        Text(
                            text = "LETRAS",
                            color = AuroColor.TextPrimary,
                            fontSize = 14.sp,
                            fontFamily = FontFamily.Monospace,
                            fontWeight = FontWeight.Bold
                        )
                        Text(
                            text = "${track.title} • ${lyrics?.source ?: "Sem letras locais"}",
                            color = AuroColor.TextTertiary,
                            fontSize = 11.5.sp,
                            maxLines = 1,
                            overflow = TextOverflow.Ellipsis
                        )
                    }
                    IconButton(onClick = onDismiss, modifier = Modifier.size(28.dp)) {
                        Icon(Icons.Rounded.Close, contentDescription = "Fechar", tint = AuroColor.TextSecondary, modifier = Modifier.size(16.dp))
                    }
                }

                HorizontalDivider(color = AuroColor.Hairline, thickness = 1.dp, modifier = Modifier.padding(vertical = 12.dp))

                if (lyrics == null || lyrics.lines.isEmpty()) {
                    Box(
                        modifier = Modifier
                            .fillMaxWidth()
                            .weight(1f),
                        contentAlignment = Alignment.Center
                    ) {
                        Column(horizontalAlignment = Alignment.CenterHorizontally) {
                            Text(
                                text = "Nenhuma letra encontrada",
                                color = AuroColor.TextPrimary,
                                fontSize = 16.sp,
                                fontWeight = FontWeight.Medium
                            )
                            Spacer(modifier = Modifier.height(6.dp))
                            Text(
                                text = "Coloque um arquivo .lrc na mesma pasta com o mesmo nome da música.",
                                color = AuroColor.TextSecondary,
                                fontSize = 12.sp,
                                textAlign = TextAlign.Center,
                                modifier = Modifier.padding(horizontal = 24.dp)
                            )
                        }
                    }
                } else {
                    LazyColumn(
                        state = listState,
                        modifier = Modifier
                            .fillMaxWidth()
                            .weight(1f),
                        contentPadding = PaddingValues(vertical = 16.dp),
                        verticalArrangement = Arrangement.spacedBy(16.dp)
                    ) {
                        itemsIndexed(lyrics.lines) { index, line ->
                            val isCurrent = index == activeLineIndex
                            Text(
                                text = line.text,
                                color = if (isCurrent) accent else AuroColor.TextTertiary,
                                fontSize = if (isCurrent) 18.sp else 14.sp,
                                fontWeight = if (isCurrent) FontWeight.Bold else FontWeight.Normal,
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .clickable {
                                        if (lyrics.isSynced) {
                                            onSeekToMs(line.timeMs)
                                        }
                                    }
                                    .padding(vertical = 2.dp)
                            )
                        }
                    }
                }
            }
        }
    }
}

 * alça de 24dp, abertura de espaço com animateItem(), rolagem automática nas bordas,
 * resposta tátil sliderTick a cada linha cruzada, snackbar "Faixa movida" com "Desfazer",
 * e ações de acessibilidade no leitor de tela ("Mover para cima" / "Mover para baixo").
 */
@Composable
fun AuroQueueScreen(
    priorityTracks: List<Track>,
    regularTracks: List<Track>,
    currentTrackId: Long,
    onBackClick: () -> Unit,
    onTrackClick: (Track, Int, Boolean) -> Unit,
    onRemoveTrack: (Track) -> Unit,
    onClearQueue: () -> Unit,
    onShuffleQueue: () -> Unit,
    onMoveItem: (section: QueueSection, from: Int, to: Int) -> Unit = { _, _, _ -> },
    listState: LazyListState = rememberLazyListState(),
    modifier: Modifier = Modifier,
    errorTrackIds: Set<Long> = emptySet(),
    showSnackbar: ((message: String, actionLabel: String?, onAction: (() -> Unit)?) -> Unit)? = null
) {
    val totalCount = priorityTracks.size + regularTracks.size
    val totalDuration = priorityTracks.sumOf { it.durationMs } + regularTracks.sumOf { it.durationMs }
    val view = LocalView.current
    val density = LocalDensity.current

    // Listas locais para renderização instantânea e reativa durante o gesto de arrasto
    var localPriorityTracks by remember(priorityTracks) { mutableStateOf(priorityTracks) }
    var localRegularTracks by remember(regularTracks) { mutableStateOf(regularTracks) }

    // Estados do Item Arrastado
    var draggingSection by remember { mutableStateOf<QueueSection?>(null) }
    var draggingInitialIndex by remember { mutableStateOf<Int?>(null) }
    var draggingCurrentIndex by remember { mutableStateOf<Int?>(null) }
    var dragOffsetY by remember { mutableFloatStateOf(0f) }
    var dragPointerY by remember { mutableFloatStateOf(0f) }

    // Sincroniza as listas locais quando a reprodução ou serviço atualiza externamente
    LaunchedEffect(priorityTracks) {
        if (draggingSection != QueueSection.PRIORITY) {
            localPriorityTracks = priorityTracks
        }
    }
    LaunchedEffect(regularTracks) {
        if (draggingSection != QueueSection.QUEUE) {
            localRegularTracks = regularTracks
        }
    }

    // Rolagem automática suave de 60fps perto das bordas superior e inferior
    LaunchedEffect(draggingSection) {
        if (draggingSection == null) return@LaunchedEffect
        while (isActive && draggingSection != null) {
            val viewportH = listState.layoutInfo.viewportSize.height
            if (viewportH > 0) {
                val edgeThreshold = with(density) { 70.dp.toPx() }
                if (dragPointerY > 0 && dragPointerY < edgeThreshold) {
                    val factor = ((edgeThreshold - dragPointerY) / edgeThreshold).coerceIn(0.2f, 1f)
                    listState.scrollBy(-18f * factor)
                } else if (dragPointerY > viewportH - edgeThreshold) {
                    val factor = ((dragPointerY - (viewportH - edgeThreshold)) / edgeThreshold).coerceIn(0.2f, 1f)
                    listState.scrollBy(18f * factor)
                }
            }
            delay(16)
        }
    }

    val itemHeightPx = with(density) { 68.dp.toPx() }
    val context = LocalContext.current
    val movedString = stringResource(R.string.queue_track_moved)
    val undoString = stringResource(R.string.undo)

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(AuroColor.Black)
    ) {
        AuroBreadcrumbHeader(
            items = listOf(
                AuroBreadcrumbItem("Biblioteca"),
                AuroBreadcrumbItem("Fila de Reprodução")
            ),
            onItemClick = { onBackClick() },
            canGoBack = true,
            onBackClick = onBackClick,
            rightActions = {
                if (totalCount > 0) {
                    IconButton(onClick = onShuffleQueue, modifier = Modifier.size(34.dp)) {
                        Icon(
                            imageVector = Icons.Rounded.Shuffle,
                            contentDescription = "Aleatório",
                            tint = AuroColor.TextSecondary,
                            modifier = Modifier.size(18.dp)
                        )
                    }
                    IconButton(onClick = onClearQueue, modifier = Modifier.size(34.dp)) {
                        Icon(
                            imageVector = Icons.Rounded.DeleteOutline,
                            contentDescription = "Limpar Fila",
                            tint = AuroColor.TextSecondary,
                            modifier = Modifier.size(18.dp)
                        )
                    }
                }
            }
        )

        if (totalCount == 0) {
            AuroEmptyState(
                icon = Icons.AutoMirrored.Rounded.QueueMusic,
                title = stringResource(R.string.empty_queue_title),
                message = stringResource(R.string.empty_queue_message),
                actionLabel = stringResource(R.string.empty_queue_action),
                onAction = onBackClick,
                modifier = Modifier.padding(bottom = 120.dp)
            )
        } else {
            // Banner informativo da fila
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 16.dp, vertical = 12.dp),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Column {
                    Text(
                        text = "FILA TOTAL",
                        color = AuroColor.TextTertiary,
                        fontSize = 11.sp,
                        fontFamily = FontFamily.Monospace
                    )
                    Spacer(modifier = Modifier.height(2.dp))
                    Text(
                        text = "$totalCount faixas • ${formatTimeMs(totalDuration)}",
                        color = AuroColor.TextPrimary,
                        fontSize = 14.sp,
                        fontWeight = FontWeight.Medium
                    )
                }

                if (localPriorityTracks.isNotEmpty()) {
                    Text(
                        text = "${localPriorityTracks.size} prioritárias",
                        color = LocalAuroAccent.current.a.value,
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Bold
                    )
                }
            }

            HorizontalDivider(color = AuroColor.Hairline, thickness = 1.dp)

            LazyColumn(
                state = listState,
                modifier = Modifier.fillMaxSize(),
                contentPadding = PaddingValues(bottom = 120.dp)
            ) {
                // Seção Prioritária (A SEGUIR)
                if (localPriorityTracks.isNotEmpty()) {
                    item(key = "header_priority") {
                        Text(
                            text = "A SEGUIR (PRIORITÁRIA)",
                            color = LocalAuroAccent.current.a.value,
                            fontSize = 11.sp,
                            fontFamily = FontFamily.Monospace,
                            fontWeight = FontWeight.Bold,
                            modifier = Modifier.padding(horizontal = 16.dp, vertical = 8.dp)
                        )
                    }

                    itemsIndexed(localPriorityTracks, key = { _, t -> "prio_${t.id}" }) { index, track ->
                        val isThisDragging = draggingSection == QueueSection.PRIORITY && draggingCurrentIndex == index

                        Box(
                            modifier = Modifier
                                .animateItem()
                                .zIndex(if (isThisDragging) 2f else 0f)
                                .graphicsLayer {
                                    if (isThisDragging) {
                                        translationY = dragOffsetY
                                        scaleX = 1.03f
                                        scaleY = 1.03f
                                    }
                                }
                                .background(if (isThisDragging) Color.White.copy(alpha = 0.06f) else Color.Transparent)
                        ) {
                            AuroQueueTrackRow(
                                track = track,
                                index = index,
                                totalCount = localPriorityTracks.size,
                                section = QueueSection.PRIORITY,
                                isPlaying = track.id == currentTrackId,
                                isPriority = true,
                                isDragging = isThisDragging,
                                hasError = errorTrackIds.contains(track.id),
                                onClick = { onTrackClick(track, index, true) },
                                onRemove = { onRemoveTrack(track) },

                                            AuroHaptics.longPress(view)
                                            draggingSection = QueueSection.PRIORITY
                                            draggingInitialIndex = index
                                            draggingCurrentIndex = index
                                            dragOffsetY = 0f
                                            dragPointerY = with(density) { (index * 68).dp.toPx() }
                                        },
                                        onDrag = { change, dragAmount ->
                                            change.consume()
                                            dragOffsetY += dragAmount.y
                                            dragPointerY += dragAmount.y

                                            val slotsShifted = (dragOffsetY / itemHeightPx).roundToInt()
                                            val currentIdx = draggingCurrentIndex ?: index
                                            val targetIdx = (currentIdx + slotsShifted).coerceIn(0, localPriorityTracks.size - 1)

                                            if (targetIdx != currentIdx) {
                                                AuroHaptics.sliderTick(view)
                                                val list = localPriorityTracks.toMutableList()
                                                val item = list.removeAt(currentIdx)
                                                list.add(targetIdx, item)
                                                localPriorityTracks = list
                                                dragOffsetY -= (targetIdx - currentIdx) * itemHeightPx
                                                draggingCurrentIndex = targetIdx
                                            }
                                        },
                                        onDragEnd = {
                                            val initial = draggingInitialIndex
                                            val target = draggingCurrentIndex
                                            if (initial != null && target != null && initial != target) {

                                                showSnackbar?.invoke(movedString, undoString) {

                                .graphicsLayer {

                                        translationY = dragOffsetY
                                        scaleX = 1.03f
                                        scaleY = 1.03f
                                    }
                                }
                                .background(if (isThisDragging) Color.White.copy(alpha = 0.06f) else Color.Transparent)
                        ) {
                            AuroQueueTrackRow(
                                track = track,
                                index = index,
                                totalCount = localRegularTracks.size,
                                section = QueueSection.QUEUE,
                                isPlaying = track.id == currentTrackId,
                                isPriority = false,
                                isDragging = isThisDragging,
                                hasError = errorTrackIds.contains(track.id),
                                onClick = { onTrackClick(track, index, false) },
                                onRemove = { onRemoveTrack(track) },
                                onAccessibilityMove = { from, to ->
                                    onMoveItem(QueueSection.QUEUE, from, to)
                                    showSnackbar?.invoke(movedString, undoString) {

@Composable
private fun AuroQueueTrackRow(
    track: Track,
    index: Int,
    totalCount: Int,
    section: QueueSection,
    isPlaying: Boolean,
    isPriority: Boolean,
    isDragging: Boolean,
    hasError: Boolean = false,
    onClick: () -> Unit,
    onRemove: () -> Unit,
    onAccessibilityMove: (from: Int, to: Int) -> Unit,
    dragHandleModifier: Modifier = Modifier
) {
    val accent = LocalAuroAccent.current.a.value
    val context = LocalContext.current
    val moveUpText = stringResource(R.string.queue_move_up)
    val moveDownText = stringResource(R.string.queue_move_down)

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clickable(onClick = onClick)
            .padding(horizontal = 16.dp, vertical = 12.dp)
            .semantics {
                if (index > 0) {
                    customActions = customActions + CustomAccessibilityAction(moveUpText) {
                        onAccessibilityMove(index, index - 1)
                        true
                    }
                }
                if (index < totalCount - 1) {
                    customActions = customActions + CustomAccessibilityAction(moveDownText) {
                        onAccessibilityMove(index, index + 1)
                        true
                    }
                }
            },
        verticalAlignment = Alignment.CenterVertically
    ) {
        val coverBitmap = rememberTrackCover(track, 100)

        if (coverBitmap != null) {
            Box(
                modifier = Modifier
                    .size(42.dp)
                    .clip(RoundedCornerShape(6.dp)),
                contentAlignment = Alignment.Center
            ) {
                Image(
                    bitmap = coverBitmap.asImageBitmap(),
                    contentDescription = null,
                    contentScale = ContentScale.Crop,
                    modifier = Modifier.fillMaxSize()
                )
                if (isPlaying) {
                    Box(
                        modifier = Modifier
                            .fillMaxSize()
                            .background(Color.Black.copy(alpha = 0.45f)),
                        contentAlignment = Alignment.Center
                    ) {
                        Text(
                            text = "ılı",
                            color = accent,
                            fontSize = 16.sp,
                            fontWeight = FontWeight.Bold
                        )
                    }
                }
            }
            Spacer(modifier = Modifier.width(12.dp))
        } else {
            Box(
                modifier = Modifier.width(28.dp),
                contentAlignment = Alignment.CenterStart
            ) {
                if (isPlaying) {
                    Text(
                        text = "ılı",
                        color = accent,
                        fontSize = 14.sp,
                        fontWeight = FontWeight.Bold
                    )
                } else {
                    Text(
                        text = if (isPriority) "★${index + 1}" else "${index + 1}",
                        color = if (isPriority) accent else AuroColor.TextTertiary,
                        fontSize = 11.sp,
                        fontFamily = FontFamily.Monospace
                    )
                }
            }
        }

        Column(modifier = Modifier.weight(1f)) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                if (hasError) {
                    Icon(
                        imageVector = Icons.Rounded.WarningAmber,
                        contentDescription = "Erro de reprodução",
                        tint = Color(0xFFFF5252),
                        modifier = Modifier.size(15.dp)
                    )
                    Spacer(modifier = Modifier.width(6.dp))
                }
                Text(
                    text = track.title,
                    color = if (hasError) Color(0xFFFF5252) else if (isPlaying) accent else AuroColor.TextPrimary,
                    fontSize = 14.sp,
                    fontWeight = if (isPlaying) FontWeight.Bold else FontWeight.Medium,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis
                )
            }
            Spacer(modifier = Modifier.height(2.dp))
            Text(
                text = track.artist.ifEmpty { "Desconhecido" },
                color = AuroColor.TextSecondary,
                fontSize = 11.5.sp,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
        }

        Spacer(modifier = Modifier.width(8.dp))

        Text(
            text = formatTimeMs(track.durationMs),
            color = AuroColor.TextSecondary,
            fontSize = 11.sp,
            fontFamily = FontFamily.Monospace
        )

        Spacer(modifier = Modifier.width(4.dp))

        IconButton(onClick = onRemove, modifier = Modifier.size(32.dp)) {
            Icon(
                imageVector = Icons.Rounded.Close,
                contentDescription = "Remover",
                tint = AuroColor.TextTertiary,
                modifier = Modifier.size(16.dp)
            )
        }

        Spacer(modifier = Modifier.width(2.dp))

        // Alça de 24dp para reordenação por arrasto (Fase 4.1)
        Box(
            modifier = Modifier
                .size(36.dp)
                .then(dragHandleModifier),
            contentAlignment = Alignment.Center
        ) {
            Icon(
                imageVector = Icons.Rounded.DragHandle,
                contentDescription = stringResource(R.string.queue_reorder_handle),
                tint = if (isDragging) accent else AuroColor.TextTertiary,
                modifier = Modifier.size(24.dp)
            )
        }
    }
}

enum class SearchFilter {

/**
 * AURŌ DESIGN SYSTEM v2 - TELA DE BUSCA (AURŌ SEARCH - FASE 3.3)
 */
@Composable
fun AuroSearchScreen(
    searchQuery: String,
    onQueryChange: (String) -> Unit,
    allTracks: List<Track>,
    currentPlayingId: Long,
    preferencesManager: PreferencesManager,
    onTrackClick: (Track, Int) -> Unit,
    onTrackMenuClick: (Track) -> Unit,
    onAlbumClick: (String) -> Unit,
    onArtistClick: (String) -> Unit,
    onClose: () -> Unit,
    errorTrackIds: Set<Long> = emptySet(),
    listState: LazyListState = rememberLazyListState(),
    modifier: Modifier = Modifier
) {
    var queryInput by remember { mutableStateOf(searchQuery) }
    var debouncedQuery by remember { mutableStateOf(searchQuery) }
    var selectedFilter by remember { mutableStateOf(SearchFilter.ALL) }
    val recentSearches = remember { mutableStateListOf<String>() }

    LaunchedEffect(Unit) {
        recentSearches.clear()
        recentSearches.addAll(preferencesManager.getRecentSearches())
    }

    // Debounce de 250ms ao digitar
    LaunchedEffect(queryInput) {
        delay(250)
        debouncedQuery = queryInput
        onQueryChange(queryInput)
    }

    // Auto-ocultar teclado ao rolar
    val keyboardController = LocalSoftwareKeyboardController.current
    LaunchedEffect(listState.isScrollInProgress) {
        if (listState.isScrollInProgress) {
            keyboardController?.hide()
        }
    }

    val trimmed = debouncedQuery.trim()
    val hasQuery = trimmed.isNotEmpty()

    // Resultados filtrados
    val matchedTracks = remember(trimmed, allTracks) {
        if (!hasQuery) emptyList()
        else allTracks.filter {
            it.title.contains(trimmed, ignoreCase = true) ||
            it.artist.contains(trimmed, ignoreCase = true) ||
            it.album.contains(trimmed, ignoreCase = true)
        }
    }

    val matchedAlbums = remember(trimmed, allTracks) {
        if (!hasQuery) emptyList()
        else allTracks.filter { it.album.contains(trimmed, ignoreCase = true) }
            .groupBy { it.album }
            .map { (albumName, tList) ->
                Album(
                    name = albumName,
                    artist = tList.firstOrNull()?.artist ?: "",
                    trackCount = tList.size,
                    year = tList.map { it.year }.filter { it > 0 }.minOrNull() ?: 0
                )
            }
            .sortedBy { it.name.lowercase() }
    }

    val matchedArtists = remember(trimmed, allTracks) {
        if (!hasQuery) emptyList()
        else allTracks.filter { it.artist.contains(trimmed, ignoreCase = true) || it.composer.contains(trimmed, ignoreCase = true) }
            .groupBy { it.artist }
            .map { (artistName, tList) ->
                Artist(
                    name = artistName,
                    trackCount = tList.size
                )
            }
            .sortedBy { it.name.lowercase() }
    }

    val hasAnyResults = matchedTracks.isNotEmpty() || matchedAlbums.isNotEmpty() || matchedArtists.isNotEmpty()

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(AuroColor.Black)
            .statusBarsPadding()
            .imePadding()
            .padding(top = 12.dp)
    ) {
        // Barra de Busca
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 14.dp, vertical = 6.dp),
            verticalAlignment = Alignment.CenterVertically
        ) {
            IconButton(onClick = onClose, modifier = Modifier.size(36.dp)) {
                Icon(
                    imageVector = Icons.AutoMirrored.Rounded.ArrowBack,
                    contentDescription = "Voltar",
                    tint = AuroColor.TextPrimary,
                    modifier = Modifier.size(20.dp)
                )
            }

            Spacer(modifier = Modifier.width(8.dp))

            // Campo de busca com vidro preto
            AuroGlass(
                kind = AuroGlassKind.Card,
                modifier = Modifier
                    .weight(1f)
                    .height(44.dp)
            ) {
                Row(
                    modifier = Modifier
                        .fillMaxSize()
                        .padding(horizontal = 12.dp),
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Icon(
                        imageVector = Icons.Rounded.Search,
                        contentDescription = null,
                        tint = AuroColor.TextTertiary,
                        modifier = Modifier.size(18.dp)
                    )
                    Spacer(modifier = Modifier.width(8.dp))
                    TextField(
                        value = queryInput,

import android.graphics.Bitmap
import androidx.compose.ui.graphics.Color
import androidx.core.graphics.ColorUtils

import kotlinx.coroutines.Dispatchers

data class AuraPalette(
    val deepBase: Color = Color(0xFF04040A),      // Obsidian Deep Black (#04040A)
    val primaryAura: Color = Color(0xFFF5881F),   // Vibrant Inner Glow (default Walk Amber)
    val secondaryAura: Color = Color(0xFF9E4708), // Secondary Transition
    val accentGlow: Color = Color(0xFFFFD875)     // Highlight Bloom
)

object AuraPaletteExtractor {

    // Paleta padrão Amber Obsidian (Estilo Walk 4,151 steps)
    val DARK_OBSIDIAN = AuraPalette(
        deepBase = Color(0xFF04040A),
        primaryAura = Color(0xFFF5881F),
        secondaryAura = Color(0xFF9E4708),
        accentGlow = Color(0xFFFFD875)
    )

    // Paleta padrão Rosa Magenta (Estilo Last test 3.3L)
    val MAGENTA_PULSE = AuraPalette(
        deepBase = Color(0xFF04040A),
        primaryAura = Color(0xFFE655A5),
        secondaryAura = Color(0xFF7D2258),
        accentGlow = Color(0xFFFFB0DC)
    )

    // Paleta padrão Laranja Coral (Estilo Breath Volume 5.5L)
    val CORAL_BREATH = AuraPalette(
        deepBase = Color(0xFF04040A),
        primaryAura = Color(0xFFE84C27),
        secondaryAura = Color(0xFF8A220C),
        accentGlow = Color(0xFFFFA878)
    )

    // Paleta padrão Verde Sálvia (Estilo Sensor Sheet)
    val SAGE_SENSOR = AuraPalette(
        deepBase = Color(0xFF04040A),
        primaryAura = Color(0xFF88AD9E),
        secondaryAura = Color(0xFF496B5E),
        accentGlow = Color(0xFFC8E6DA)
    )

    // Paleta padrão Azul Cobalto (Estilo Training)
    val COBALT_TRAINING = AuraPalette(
        deepBase = Color(0xFF04040A),
        primaryAura = Color(0xFF2848B8),
        secondaryAura = Color(0xFF102060),
        accentGlow = Color(0xFF7898FF)
    )

    suspend fun extractFromBitmapAsync(bitmap: Bitmap?): AuraPalette = withContext(Dispatchers.Default) {
        extractFromBitmap(bitmap)
    }

    fun extractFromBitmap(bitmap: Bitmap?): AuraPalette {
        if (bitmap == null || bitmap.isRecycled) return DARK_OBSIDIAN

        // Downscale obrigatório para 64x64 em background para evitar qualquer ANR na Main Thread
        val scaled = try {
            Bitmap.createScaledBitmap(bitmap, 64, 64, true)
        } catch (_: Exception) {
            return DARK_OBSIDIAN
        }

        val palette = Palette.from(scaled).maximumColorCount(16).generate()
        val vibrant = palette.vibrantSwatch ?: palette.lightVibrantSwatch ?: palette.darkVibrantSwatch
        val dominant = palette.dominantSwatch

        val candidateRgb = vibrant?.rgb ?: dominant?.rgb ?: return DARK_OBSIDIAN
        val hsl = FloatArray(3)
        ColorUtils.colorToHSL(candidateRgb, hsl)

        // Se a capa for em tons de cinza ou monocromática, mantém Dark Obsidian
        if (hsl[1] < 0.20f) {
            return DARK_OBSIDIAN
        }

        // Garante saturação rica para aura luminosa
        val primaryHsl = floatArrayOf(hsl[0], hsl[1].coerceIn(0.65f, 0.95f), hsl[2].coerceIn(0.45f, 0.62f))
        val secondaryHsl = floatArrayOf((hsl[0] + 25f) % 360f, (hsl[1] * 0.85f).coerceIn(0.50f, 0.85f), 0.30f)
        val glowHsl = floatArrayOf((hsl[0] + 345f) % 360f, 0.88f, 0.78f)

        return AuraPalette(
            deepBase = Color(0xFF04040A),
            primaryAura = Color(ColorUtils.HSLToColor(primaryHsl)),
            secondaryAura = Color(ColorUtils.HSLToColor(secondaryHsl)),
            accentGlow = Color(ColorUtils.HSLToColor(glowHsl))
        )
    }
}

// <GlowTone>
/** Cor-semente da capa -> duas cores de glow que brilham sobre o preto. Pura (sem Android), testável na JVM. */
object GlowTone {
    fun fit(argb: Int): Pair<Int, Int> {
        val r = ((argb shr 16) and 0xFF) / 255f
        val g = ((argb shr 8) and 0xFF) / 255f
        val b = (argb and 0xFF) / 255f
        val hsl = rgbToHsl(r, g, b)
        val h = hsl[0]; val s = hsl[1]; val l = hsl[2]
        return if (s < 0.10f) {
            // capa sem cor (cinza/preto/branco): glow prata quente
            hslToArgb(35f / 360f, 0.08f, 0.80f) to hslToArgb(35f / 360f, 0.08f, 0.62f)
        } else {
            val s1 = (s * 1.15f).coerceIn(0.62f, 0.95f)
            val l1 = l.coerceIn(0.50f, 0.62f)
            val h2 = (h + 28f / 360f) % 1f // vizinha análoga, um pouco mais clara
            hslToArgb(h, s1, l1) to hslToArgb(h2, (s1 - 0.05f).coerceIn(0f, 1f), (l1 + 0.06f).coerceIn(0f, 0.7f))
        }
    }

    fun rgbToHsl(r: Float, g: Float, b: Float): FloatArray {
        val mx = maxOf(r, g, b); val mn = minOf(r, g, b)
        val l = (mx + mn) / 2f
        if (mx == mn) return floatArrayOf(0f, 0f, l)
        val d = mx - mn
        val s = if (l > 0.5f) d / (2f - mx - mn) else d / (mx + mn)
        val h = when (mx) {
            r -> ((g - b) / d + (if (g < b) 6f else 0f)) / 6f
            g -> ((b - r) / d + 2f) / 6f
            else -> ((r - g) / d + 4f) / 6f
        }
        return floatArrayOf(h, s, l)
    }

    private fun hue(p: Float, q: Float, tIn: Float): Float {
        var t = tIn
        if (t < 0f) t += 1f
        if (t > 1f) t -= 1f
        return when {
            t < 1f / 6f -> p + (q - p) * 6f * t
            t < 1f / 2f -> q
            t < 2f / 3f -> p + (q - p) * (2f / 3f - t) * 6f
            else -> p
        }
    }

    fun hslToArgb(h: Float, s: Float, l: Float): Int {
        val r: Float; val g: Float; val b: Float
        if (s == 0f) { r = l; g = l; b = l } else {
            val q = if (l < 0.5f) l * (1f + s) else l + s - l * s
            val p = 2f * l - q
            r = hue(p, q, h + 1f / 3f); g = hue(p, q, h); b = hue(p, q, h - 1f / 3f)
        }
        fun c(v: Float) = (v * 255f + 0.5f).toInt().coerceIn(0, 255)
        return (0xFF shl 24) or (c(r) shl 16) or (c(g) shl 8) or c(b)
    }
}
// </GlowTone>

object CoverGlow {
    private val memoryCache = mutableMapOf<Int, AuroAccent>()

    /** Capa -> cores de glow com cache. Quantiza a imagem: chame fora da main thread. */
    suspend fun from(cover: Bitmap?): AuroAccent = withContext(Dispatchers.Default) {
        if (cover == null) return@withContext AuroAccent.Neutral

        val cacheKey = cover.hashCode()
        memoryCache[cacheKey]?.let { return@withContext it }

        try {
            // Conversão segura caso o bitmap seja Hardware
            val swBitmap = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O && cover.config == Bitmap.Config.HARDWARE) {
                cover.copy(Bitmap.Config.ARGB_8888, false) ?: return@withContext AuroAccent.Neutral
            } else {
                cover
            }

            val p = Palette.from(swBitmap).maximumColorCount(16).resizeBitmapArea(96 * 96).generate()
            val swatches = listOfNotNull(p.vibrantSwatch, p.lightVibrantSwatch, p.darkVibrantSwatch, p.dominantSwatch)
            val seed = swatches.maxByOrNull { it.population * (0.25f + it.hsl[1]) }
                ?: return@withContext AuroAccent.Neutral
            val (a, b) = GlowTone.fit(seed.rgb)
            val result = AuroAccent(Color(a), Color(b))
            if (memoryCache.size > 100) memoryCache.clear()
            memoryCache[cacheKey] = result
            result
        } catch (_: Exception) {
            AuroAccent.Neutral
        }
    }
}

// ─────────────────── Luz do ouvinte: inclinação do celular ───────────────────

// <AuroTilt>
/** Inclinação do celular -> direção da luz do ouvinte. Pura (sem Android), testável na JVM. */
object AuroTilt {
    /**
     * gx, gy: gravidade nos eixos do aparelho, em m/s² (convenção do Android: x pra direita, y pra cima na tela;
     * com o celular em pé, gy ≈ +9,81). A luz vem do "pra baixo do mundo" projetado na tela, com um viés pro pé
     * da tela (o ouvinte) quando o celular está deitado. Retorna (dx, dy, ganho): direção PARA a luz
     * (x pra direita, y pra baixo) e intensidade 0,6..1 (celular deitado = reflexo um pouco mais fraco).
     */
    fun light(gx: Float, gy: Float, sideGain: Float = 1.6f, bias: Float = 0.25f): FloatArray {
        val x = -gx / 9.81f * sideGain
        val y = gy / 9.81f + bias
        val len = sqrt(x * x + y * y)
        val inPlane = sqrt(gx * gx + gy * gy) / 9.81f
        val gain = 0.6f + 0.4f * inPlane.coerceIn(0f, 1f)
        return if (len < 1e-4f) floatArrayOf(0f, 1f, gain) else floatArrayOf(x / len, y / len, gain)
    }
}

/** Mola amortecida (ω = 7 rad/s, ζ = 0,8): a luz acompanha o celular com um pouco de atraso, como líquido. */
class TiltSmoother(private val omega: Float = 7f, private val zeta: Float = 0.8f) {
    var x = 0f
        private set
    var y = 1f
        private set
    var gain = 1f
        private set
    private var vx = 0f
    private var vy = 0f
    private var vg = 0f
    private var last = 0L

    fun step(tx: Float, ty: Float, tg: Float, nanos: Long) {
        val dt = if (last == 0L) 0.02f else ((nanos - last) / 1e9f).coerceIn(0.001f, 0.05f)
        last = nanos
        vx += (omega * omega * (tx - x) - 2f * zeta * omega * vx) * dt
        x += vx * dt
        vy += (omega * omega * (ty - y) - 2f * zeta * omega * vy) * dt
        y += vy * dt
        vg += (omega * omega * (tg - gain) - 2f * zeta * omega * vg) * dt
        gain += vg * dt
    }
}
// </AuroTilt>

/**
 * Direção PARA a luz no plano da tela (x pra direita, y pra baixo): (0, 1) = vem de baixo. gain = 0,6..1.
 * Só existe UMA luz, igual para todos os objetos.
 */
@Immutable
data class AuroLight(val dx: Float = 0f, val dy: Float = 1f, val gain: Float = 1f)

class AuroAccentState(val a: State<Color>, val b: State<Color>)

// Leitores só em fase de desenho (lambda / State): a animação NÃO recompõe a tela a cada frame.
val LocalAuroLight = staticCompositionLocalOf<() -> AuroLight> { { AuroLight() } }
val LocalAuroTime = staticCompositionLocalOf<() -> Float> { { 0f } }
val LocalAuroAccent = staticCompositionLocalOf<AuroAccentState> {
    AuroAccentState(
        object : State<Color> { override val value = AuroAccent.Neutral.a },
        object : State<Color> { override val value = AuroAccent.Neutral.b },
    )
}

val LocalAuroSoft = staticCompositionLocalOf { 0.6f }
val LocalReduceMotionOverride = staticCompositionLocalOf<Boolean?> { null }

/** true se o usuário desligou as animações do sistema ou configurou no app. */
@Composable
fun rememberReduceMotion(): Boolean {
    val override = LocalReduceMotionOverride.current
    if (override != null) return override
    val context = LocalContext.current
    return remember { Settings.Global.getFloat(context.contentResolver, Settings.Global.ANIMATOR_DURATION_SCALE, 1f) == 0f }
}

/**
 * Luz do ouvinte: vem de baixo e acompanha o celular (sensor de gravidade, suavizado por mola). Enquanto a
 * música toca, a direção respira ±4° devagar. Sem sensor ou com "remover animações", fica fixa embaixo.
 */
@Composable
fun rememberListenerLight(playing: Boolean, tilt: Boolean = true): () -> AuroLight {
    val context = LocalContext.current
    val sensed = remember { mutableStateOf(AuroLight()) } // lido só na fase de desenho: não recompõe
    DisposableEffect(tilt) {
        val sm = context.getSystemService(Context.SENSOR_SERVICE) as SensorManager
        val sensor = sm.getDefaultSensor(Sensor.TYPE_GRAVITY) ?: sm.getDefaultSensor(Sensor.TYPE_ACCELEROMETER)
        if (!tilt || sensor == null) {
            sensed.value = AuroLight()
            return@DisposableEffect onDispose {}
        }
        val smoother = TiltSmoother()
        val listener = object : SensorEventListener {
            override fun onSensorChanged(event: SensorEvent) {
                val l = AuroTilt.light(event.values[0], event.values[1])
                smoother.step(l[0], l[1], l[2], event.timestamp)
                val len = sqrt(smoother.x * smoother.x + smoother.y * smoother.y).coerceAtLeast(1e-4f)
                val next = AuroLight(smoother.x / len, smoother.y / len, smoother.gain)
                val cur = sensed.value
                if (abs(next.dx - cur.dx) + abs(next.dy - cur.dy) + abs(next.gain - cur.gain) > 0.002f) sensed.value = next
            }

            override fun onAccuracyChanged(s: Sensor?, accuracy: Int) {}
        }
        sm.registerListener(listener, sensor, SensorManager.SENSOR_DELAY_GAME)
        onDispose { sm.unregisterListener(listener) }
    }

    val transition = rememberInfiniteTransition(label = "light")
    val phase = transition.animateFloat(
        initialValue = 0f,
        targetValue = 1f,
        animationSpec = infiniteRepeatable(tween(AuroMotion.DriftMs, easing = LinearEasing)),
        label = "phase",
    )
    val amount = animateFloatAsState(if (playing && tilt) 1f else 0f, AuroMotion.liquid(), label = "breath")
    return {
        val s = sensed.value
        val th = 0.07f * amount.value * sin((phase.value * 2f * PI).toFloat()) // ±4°
        val c = cos(th)
        val sn = sin(th)
        AuroLight(dx = s.dx * c - s.dy * sn, dy = s.dx * sn + s.dy * c, gain = s.gain)
    }
}

/** Relógio em segundos para as ondas lentas da luz. Só corre enquanto toca (economiza bateria). */
@Composable
fun rememberAuroClock(running: Boolean): () -> Float {
    val t = produceState(0f, running) {
        if (!running) return@produceState
        val start = withFrameNanos { it }
        val offset = value
        while (true) withFrameNanos { value = offset + (it - start) / 1_000_000_000f }
    }
    return { t.value }
}

@Composable
fun AuroTheme(
    accent: AuroAccent,
    playing: Boolean = true,
    tilt: Boolean = true,
    soft: Float = 0.6f,
    reduceMotionOverride: Boolean? = null,
    content: @Composable () -> Unit
) {
    val reduce = if (reduceMotionOverride != null) reduceMotionOverride else rememberReduceMotion()
    val a = animateColorAsState(accent.a, if (reduce) tween(0) else AuroMotion.accent(), label = "accentA")
    val b = animateColorAsState(accent.b, if (reduce) tween(0) else AuroMotion.accent(), label = "accentB")
    val state = remember(a, b) { AuroAccentState(a, b) }
    val light = rememberListenerLight(playing && !reduce, tilt && !reduce)
    val clock = rememberAuroClock(playing && !reduce)
    CompositionLocalProvider(
        LocalAuroAccent provides state,
        LocalAuroLight provides light,
        LocalAuroTime provides clock,
        LocalAuroSoft provides soft,
        LocalReduceMotionOverride provides reduceMotionOverride
    ) {
        Box(Modifier.fillMaxSize().background(AuroColor.Black)) { content() }
    }
}

// ─────────────────────────── Objeto de vidro ───────────────────────────

/** Button = círculo (anterior, play, próximo, ★, ⋮). Card = retângulo arredondado (Fila, Letras, mini player). */
enum class AuroGlassKind { Button, Card }

@Composable
fun AuroGlass(
    modifier: Modifier = Modifier,
    kind: AuroGlassKind = AuroGlassKind.Card,
    onClick: (() -> Unit)? = null,
    haptic: ((View) -> Unit)? = { AuroHaptics.click(it) },
    soft: Float = LocalAuroSoft.current, // 0 = reflexo agudo, 1 = bem borrado
    content: @Composable BoxScope.() -> Unit,
) {
    val density = LocalDensity.current.density
    val cfg = LocalConfiguration.current
    val view = LocalView.current
    val screenH = cfg.screenHeightDp * density
    var center by remember { mutableStateOf(Offset.Zero) } // centro do objeto na janela, px
    val interaction = remember { MutableInteractionSource() }
    val pressed by interaction.collectIsPressedAsState()
    val press by animateFloatAsState(if (pressed) 1f else 0f, AuroMotion.press(), label = "press")
    val isButton = kind == AuroGlassKind.Button
    val shape = if (isButton) CircleShape else RoundedCornerShape(AuroRadius.card)

    val clickModifier = if (onClick != null) {
        Modifier.clickable(interaction, indication = null) {
            haptic?.invoke(view)
            onClick()
        }
    } else Modifier

    if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
        val shader = remember { RuntimeShader(AURO_GLASS_AGSL) }
        val brush = remember(shader) { ShaderBrush(shader) }
        val light = LocalAuroLight.current
        val clock = LocalAuroTime.current
        val accent = LocalAuroAccent.current

        Box(
            modifier
                .onGloballyPositioned { center = it.boundsInWindow().center }
                .graphicsLayer { val s = 1f - 0.04f * press; scaleX = s; scaleY = s }
                .drawBehind {
                    val l = light()
                    val radius = if (isButton) min(size.width, size.height) / 2f else AuroRadius.card.toPx()
                    val bevel = if (isButton) 0.34f * radius else 13.dp.toPx()
                    val a = accent.a.value
                    val b = accent.b.value
                    shader.setFloatUniform("uSize", size.width, size.height)
                    shader.setFloatUniform("uRadius", radius)
                    shader.setFloatUniform("uCenter", center.x, center.y)
                    shader.setFloatUniform("uScreenH", screenH)
                    shader.setFloatUniform("uLightDir", l.dx, l.dy)
                    shader.setFloatUniform("uAccent", a.red, a.green, a.blue)
                    shader.setFloatUniform("uAccent2", b.red, b.green, b.blue)
                    shader.setFloatUniform("uBevel", bevel)
                    shader.setFloatUniform("uSoft", soft)
                    shader.setFloatUniform("uSpec", (if (isButton) 1.5f else 1.3f) * l.gain)
                    shader.setFloatUniform("uBody", 0f) // 0 = corpo de vidro preto total
                    shader.setFloatUniform("uTime", clock())
                    shader.setFloatUniform("uPx", density)
                    shader.setFloatUniform("uPress", press)
                    drawRect(brush)
                }
                .clip(shape)
                .then(clickModifier),
            content = content,
        )
    } else {
        // Fallback robusto para Android < 13: Objeto preto com aro branco sutil de 10%
        Box(
            modifier = modifier
                .graphicsLayer { val s = 1f - 0.04f * press; scaleX = s; scaleY = s }
                .background(AuroColor.Black, shape)
                .border(1.dp, Color(0x1AFFFFFF), shape)
                .clip(shape)
                .then(clickModifier),
            content = content,
        )
    }
}

// Shader AGSL do vidro preto: luz paralela do ouvinte refletida nas bordas.
private const val AURO_GLASS_AGSL = """
// Aurō Glass v3: vidro preto sobre o vazio (#000000), iluminado pela luz do OUVINTE:
// uma luz paralela vinda de baixo (igual em todos os objetos) que acompanha a inclinação do celular.
// A luz não é desenhada: aparece só como reflexo nas bordas. AGSL, Android 13+.

uniform float2 uSize;      // px do objeto
uniform float  uRadius;    // px do raio do canto (círculo: metade do lado)
uniform float2 uCenter;    // centro do objeto na janela, px (ondas lentas e gradiente vertical)
uniform float  uScreenH;   // altura da tela, px
uniform float2 uLightDir;  // direção PARA a luz no plano da tela (y pra baixo): (0,1) = vem de baixo
uniform float3 uAccent;    // cor da luz (vem da capa)
uniform float3 uAccent2;   // cor secundária (vem da capa)
uniform float  uBevel;     // px: largura do chanfro (usada só no reflexo interno fraco)
uniform float  uSoft;      // 0 = reflexo agudo, 1 = bem borrado
uniform float  uSpec;      // intensidade do reflexo
uniform float  uBody;      // luz no miolo (0 = preto total)
uniform float  uTime;      // s
uniform float  uPx;        // 1dp em px
uniform float  uPress;     // 0..1 (toque)

float sdRound(float2 p) {
    float2 q = abs(p) - 0.5 * uSize + uRadius;
    return min(max(q.x, q.y), 0.0) + length(max(q, 0.0)) - uRadius;
}

float hash2(float2 p) {
    float3 p3 = fract(float3(p.x, p.y, p.x) * 0.1031);
    p3 += dot(p3, p3.yzx + 33.33);
    return fract((p3.x + p3.y) * p3.z);
}

// Intensidade da luz neste ponto: um pouco mais forte embaixo (perto do ouvinte), com ondas lentas.
float lightI(float2 g) {
    float yN = clamp(g.y / uScreenH, 0.0, 1.0);
    float base = mix(0.6, 1.0, yN);
    float flux = 1.0 + 0.05 * sin(1.3 * uTime) + 0.03 * sin(2.9 * uTime + 1.7);
    float2 gd = g / uPx;                                   // em dp
    float wave = sin(gd.x * 0.055 + 1.4 * sin(gd.y * 0.04 + uTime * 0.45) + uTime * 0.3);
    return base * flux * (1.0 + 0.12 * wave);
}

half4 main(float2 fragCoord) {
    float2 p = fragCoord - 0.5 * uSize;
    float d = sdRound(p);
    float a = 1.0 - smoothstep(-0.7, 0.7, d);
    if (a <= 0.0) return half4(0.0);
    float depth = max(-d, 0.0);
    float dd = depth / uPx;                                 // profundidade em dp: a espessura das linhas é igual em todos os objetos
    float2 n = normalize(float2(sdRound(p + float2(1.0, 0.0)) - sdRound(p - float2(1.0, 0.0)),
                                sdRound(p + float2(0.0, 1.0)) - sdRound(p - float2(0.0, 1.0))) + float2(1e-5));
    float2 g = uCenter + p;
    float I = lightI(g);
    float press = 1.0 + 0.6 * uPress;

    // luz paralela: a MESMA direção em todos os objetos. Dispersão por canal: a direção gira ±2°.
    float2 L = normalize(uLightDir);
    float2 Lp = float2(-L.y, L.x);
    float3 ndl = float3(max(dot(n, L * 0.99939 + Lp * 0.03490), 0.0),
                        max(dot(n, L), 0.0),
                        max(dot(n, L * 0.99939 - Lp * 0.03490), 0.0));

    // reflexo: um lóbulo estreito e um largo; uSoft controla o quão borrado
    float eN = mix(10.0, 4.0, uSoft);
    float eW = mix(3.0, 1.4, uSoft);
    float3 lN = pow(ndl, float3(eN));
    float3 lW = pow(ndl, float3(eW));
    float core = exp(-dd / 1.0);                            // linha de ~1dp
    float mid = exp(-dd / (2.2 + 2.6 * uSoft));             // faixa de poucos dp
    float bloom = exp(-dd / (5.0 + 6.0 * uSoft));           // brilho suave que entra no vidro
    float3 hot = mix(uAccent, float3(1.0), 0.55);
    float3 refl = hot * (lN * (0.55 * core + 0.55 * mid) + lW * (0.30 * mid + 0.10 * bloom)) * I;
    float3 sparkCol = mix(hot, float3(1.0), 0.6) * lN.g * core * mix(0.9, 0.5, uSoft) * I;

    // linha de contorno com presença: contínua em volta do objeto, mais forte do lado da luz
    float3 rimCol = mix(float3(1.0), uAccent, 0.25) * exp(-dd / 1.0) * (0.06 + 0.20 * pow(ndl.g, 1.5)) * I;

    // a luz que atravessa sai pelo lado oposto: fio interno fraco
    float opp = pow(max(-dot(n, L), 0.0), 3.0);
    float band = exp(-pow((depth - 0.5 * uBevel) / (0.3 * uBevel), 2.0));
    float3 inner = mix(uAccent2, hot, 0.4) * opp * band * 0.07 * I;

    float3 body = uAccent * uBody * I * exp(-dd / 6.0);     // 0 = corpo preto

    float3 c = body + (refl + sparkCol) * uSpec * press + rimCol * press + inner;
    float mx = max(c.r, max(c.g, c.b));
    c = max(c + (hash2(fragCoord) - 0.5) * (1.6 / 255.0) * smoothstep(0.0, 0.03, mx), 0.0);   // dither só onde há luz
    return half4(half3(clamp(c, 0.0, 1.0) * a), half(a));
}
"""
